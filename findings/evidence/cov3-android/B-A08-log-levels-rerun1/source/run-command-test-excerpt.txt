            XCTAssertEqual(RunCommand.exitCode(for: error), .validationFailure)

            XCTAssertTrue(message.contains("warning"), message)
            XCTAssertTrue(message.contains("default") && message.contains("info") && message.contains("debug"), message)
        }
    }
